    <script>
        const wordBank = ["יריחו", "רחב", "ירדן", "חומות", "עכן", "העי", "גבעונים", "שמש", "אבנים", "גלגל", "עתניאל", "אהוד", "עגלון", "דבורה", "ברק", "סיסרא", "יעל", "יתד", "גדעון", "כדים", "יפתח", "שמשון", "דלילה", "אריה", "פלשתים", "עלי", "חנה", "פנינה", "שמואל", "ארון הברית", "דגון", "שאול", "אתונות", "קיש", "יונתן", "גלית", "קלע", "דוד", "מיכל", "כנור", "עין גדי", "בעלת האוב", "גלבוע"];
        let cardColors = [], cardRevealed = Array(25).fill(false), viewMode = 'classroom', timeLeft = 60, isTimerRunning = false, isMusicPlaying = false;

        // --- STREAMING_CHUNK:Implementing Logic ---
        function hashString(str) { let hash = 0; for(let i=0; i<str.length; i++) hash = ((hash<<5)-hash)+str.charCodeAt(i); return Math.abs(hash); }
        function seededRandom(seed) { return function() { seed = Math.imul(seed ^ seed >>> 15, seed | 1); return (seed >>> 0) / 4294967296; } }
        
        function generateGame(code) {
            let rng = seededRandom(hashString(code));
            let shuffled = [...wordBank].sort(() => rng() - 0.5);
            let colors = [...Array(9).fill('red'), ...Array(8).fill('blue'), ...Array(7).fill('neutral'), 'assassin'].sort(() => rng() - 0.5);
            cardColors = colors; cardRevealed = Array(25).fill(false);
            renderBoard();
        }

        function renderBoard() {
            const board = document.getElementById('gameBoard'); board.innerHTML = '';
            for (let i = 0; i < 25; i++) {
                const btn = document.createElement('button');
                btn.className = `card h-24 rounded-xl font-bold p-2 ${viewMode === 'classroom' && cardRevealed[i] ? 'card-' + cardColors[i] : 'card-hidden'}`;
                btn.innerText = wordBank[i];
                if (viewMode === 'classroom') btn.onclick = () => reveal(i);
                else if (viewMode === 'spymaster') btn.classList.add('card-' + cardColors[i]);
                board.appendChild(btn);
            }
        }

        function reveal(i) {
            cardRevealed[i] = true;
            if (cardColors[i] === 'assassin') { document.body.classList.add('shake-screen'); playEmoji('💥'); }
            else if (cardColors[i] === 'neutral') playEmoji('😢');
            else playEmoji('🎉');
            renderBoard();
        }

        function playEmoji(e) { 
            const o = document.getElementById('emojiOverlay'); o.innerText = e; o.classList.add('show-emoji'); 
            setTimeout(() => o.classList.remove('show-emoji'), 1000); 
        }

        function toggleInstructions() { document.getElementById('instructionsModal').classList.toggle('hidden'); }
        function createNewGame() { generateGame(Math.floor(Math.random()*10000).toString()); }
        function joinGame() { generateGame(document.getElementById('roomCodeInput').value); }
        function setViewMode(m) { viewMode = m; renderBoard(); }
        function toggleTimer() { /* logic here */ }
        function resetTimer() { timeLeft = 60; document.getElementById('timerDisplay').innerText = "01:00"; }
        window.onload = createNewGame;
    </script>
